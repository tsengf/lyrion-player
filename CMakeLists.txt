option(PLAYER_FETCH_FLAC "Download pinned FLAC if it is not installed" ON)
option(PLAYER_AAC "Enable AAC decoding (requires faad2)" OFF)
find_package(Threads REQUIRED)

add_library(player_codecs INTERFACE)
function(player_dependency header library)
  string(MAKE_C_IDENTIFIER "${library}" key)
  find_path(${key}_INCLUDE NAMES ${header} REQUIRED)
  find_library(${key}_LIBRARY NAMES ${library} REQUIRED)
  target_include_directories(player_codecs INTERFACE "${${key}_INCLUDE}")
  target_link_libraries(player_codecs INTERFACE "${${key}_LIBRARY}")
endfunction()
find_path(FLAC_INCLUDE NAMES FLAC/stream_decoder.h)
find_library(FLAC_LIBRARY NAMES FLAC)
if(FLAC_INCLUDE AND FLAC_LIBRARY)
  target_include_directories(player_codecs INTERFACE "${FLAC_INCLUDE}")
  target_link_libraries(player_codecs INTERFACE "${FLAC_LIBRARY}")
elseif(PLAYER_FETCH_FLAC)
  include(FetchContent)
  set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(WITH_OGG OFF CACHE BOOL "" FORCE)
  set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(flac GIT_REPOSITORY https://github.com/xiph/flac.git
    GIT_TAG 1.5.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(flac)
  target_link_libraries(player_codecs INTERFACE FLAC)
else()
  message(FATAL_ERROR "Install FLAC or enable PLAYER_FETCH_FLAC")
endif()

function(player_target name)
  add_executable(${name} ${PLAYER_SOURCES})
  target_include_directories(${name} PRIVATE ${SL} src vendor/miniaudio)
  target_compile_definitions(${name} PRIVATE PLAYER_NATIVE=1 PORTAUDIO=1 LINKALL=1
    RESAMPLE=1 NO_MAD=1 MODEL_NAME=Player)
  if(NOT PLAYER_AAC)
    target_compile_definitions(${name} PRIVATE NO_FAAD=1)
  endif()
  target_link_libraries(${name} PRIVATE player_codecs Threads::Threads)
  if(MSVC)
    target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS)
    target_link_libraries(${name} PRIVATE ws2_32 iphlpapi ole32)
  else()
    target_compile_options(${name} PRIVATE -fcommon -Wall -Wextra -Wno-unused-parameter)
    target_link_libraries(${name} PRIVATE m)
  endif()
endfunction()
player_target(player)
if(WIN32)
  target_sources(player PRIVATE src/windows_entry.c)
  target_link_libraries(player PRIVATE shell32)
  set_target_properties(player PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
install(TARGETS player RUNTIME DESTINATION bin)

option(PLAYER_BUILD_TRAY "Build the native instance-management tray application" OFF)
if(PLAYER_BUILD_TRAY)
  enable_language(CXX)
  if(MSVC)
    add_compile_options(/utf-8)
  endif()
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  add_library(player_tray_core STATIC src/tray/core.cpp src/discovery.c)
  target_include_directories(player_tray_core PUBLIC src/tray vendor/json PRIVATE src)
  target_link_libraries(player_tray_core PUBLIC Threads::Threads)
  if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    add_executable(lyrion-player-tray MACOSX_BUNDLE src/tray/mac_app.mm src/tray/platform_mac.mm)
    target_compile_options(lyrion-player-tray PRIVATE -fobjc-arc)
    target_link_libraries(lyrion-player-tray PRIVATE player_tray_core "-framework AppKit" "-framework ServiceManagement")
    set_target_properties(lyrion-player-tray PROPERTIES
      OUTPUT_NAME "Lyrion Player"
      OSX_DEPLOYMENT_TARGET "13.0"
      MACOSX_BUNDLE_GUI_IDENTIFIER "org.lyrion-player.tray"
      MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/tray/Info.plist.in")
    # Refresh the bundled executable when only backend sources change.
    set_property(TARGET lyrion-player-tray APPEND PROPERTY LINK_DEPENDS "$<TARGET_FILE:player>")
    add_custom_command(TARGET lyrion-player-tray POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:player>
        "$<TARGET_BUNDLE_CONTENT_DIR:lyrion-player-tray>/MacOS/player")
    install(TARGETS lyrion-player-tray BUNDLE DESTINATION . COMPONENT Tray)
    install(CODE [[
      include(BundleUtilities)
      set(tray_app "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/Lyrion Player.app")
      fixup_bundle("${tray_app}" "${tray_app}/Contents/MacOS/player" "")
      execute_process(COMMAND /usr/bin/codesign --force --deep --sign - "${tray_app}" RESULT_VARIABLE signing)
      if(NOT signing EQUAL 0)
        message(FATAL_ERROR "Ad-hoc signing failed")
      endif()
    ]] COMPONENT Tray)
  else()
    target_link_libraries(player_tray_core PUBLIC ws2_32 iphlpapi)
    target_compile_definitions(player_tray_core PRIVATE _CRT_SECURE_NO_WARNINGS)
    add_executable(lyrion-player-tray WIN32 src/tray/windows_app.cpp src/tray/platform_windows.cpp)
    target_compile_definitions(lyrion-player-tray PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(lyrion-player-tray PRIVATE player_tray_core comctl32 shell32 user32 gdi32 advapi32 ole32 uuid)
    set_target_properties(lyrion-player-tray PROPERTIES OUTPUT_NAME "Lyrion Player")
    install(TARGETS lyrion-player-tray player RUNTIME_DEPENDENCY_SET tray_runtime
      RUNTIME DESTINATION . COMPONENT Tray)
    install(RUNTIME_DEPENDENCY_SET tray_runtime
      DIRECTORIES "$<TARGET_FILE_DIR:player>"
      PRE_EXCLUDE_REGEXES "api-ms-.*" "ext-ms-.*"
      POST_EXCLUDE_REGEXES ".*[\\/]([Ww]indows)[\\/]([Ss]ystem32)[\\/].*"
      RUNTIME DESTINATION . COMPONENT Tray)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT Tray)
    include(InstallRequiredSystemLibraries)
  endif()
  target_compile_definitions(lyrion-player-tray PRIVATE PLAYER_APP_VERSION="${PROJECT_VERSION}")
  add_dependencies(lyrion-player-tray player)
  install(FILES LICENSE COPYING THIRD_PARTY.md vendor/json/LICENSE.MIT DESTINATION licenses COMPONENT Tray)
  install(DIRECTORY vendor/licenses/ DESTINATION licenses/codecs COMPONENT Tray)
  install(FILES vendor/miniaudio/miniaudio.h DESTINATION licenses/miniaudio COMPONENT Tray)
endif()
